// ====== 核心代码（Promise + async/await） ======

// 工具函数：把 setTimeout 包成 Promise
function wait(ms, label) {
  return new Promise(function(resolve) {
    setTimeout(function() { resolve(label); }, ms);
  });
}

async function makeDumplings() {
  // ① Promise.race —— 两个都跑，任一完成即可
  let buy = await Promise.race([
    wait(4000, '线上下单'),
    wait(6000, '线下菜市场')
  ]);

  // ② 取原料 —— 串行
  await wait(1500, '取原料');

  // ③ Promise.all —— 两个都要完成才往下
  await Promise.all([
    wait(3000, '和面'),
    wait(5000, '和馅')
  ]);

  // ④ 包饺子 → 煮饺子 → 吃饺子 都是串行
  await wait(8000, '包饺子');
  await wait(4000, '煮饺子');
  await wait(2000, '吃饺子');
}
