① if / else 条件判断 分支结构
let age = +document.getElementById('age').value;
if (age >= 18) {
  output = '已成年，可以进入网吧';
} else if (age >= 12) {
  output = '青少年，需要家长陪同';
} else {
  output = '儿童，禁止进入';

② switch 多分支选择 分支结构
let day = ['日','一','二','三','四','五','六'][new Date().getDay()];
switch (day) {
  case '一': case '二': case '三':
    msg = '新的一周，加油！'; break;
  case '五':
    msg = '周五啦，准备下班！'; break;
  case '六': case '日':
    msg = '周末愉快 🎉'; break;
  default:
    msg = '周四，再坚持一下';
}

④ while 循环（掷骰子直到 6 点）循环结构
let count = 0, dice = 0;
while (dice !== 6) {
  dice = Math.floor(Math.random()*6)+1;
  count++;
  history += '第'+count+'次：掷出 '+dice+' ⬇\n';
}
history += '终于掷出 6 点！共 '+count+' 次';

⑤ do...while 循环（至少执行一次）循环结构
// do...while 无论条件如何，循环体至少跑一次
let i = 5, text = '';
do {
  text += 'i = ' + i + '\n';
  i++;
} while (i < 3);  // 条件假，但体已执行一次

⑦ 嵌套 for（九九乘法表）循环结构
let table = '';
for (let i = 1; i <= 9; i++) {
  for (let j = 1; j <= i; j++) {
    table += j + '×' + i + '=' + (i*j) + '\t';
  }
  table += '\n';
}

⑧ break & continue（跳过/跳出）控制语句
// continue：跳过本次迭代
// break：直接终止循环
let text = '';
for (let i = 1; i <= 15; i++) {
  if (i === 12) break;          // 到 12 就停
  if (i % 3 === 0) continue;    // 跳过 3 的倍数
  text += i + ' ';
}
